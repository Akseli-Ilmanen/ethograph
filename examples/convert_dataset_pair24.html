Social rats - PAIR-R24M#

Data from Marshall et al. [2021] (figshare), a multi-animal 3D pose dataset about the dyadic interactions in laboratory rats.

Figure taken from Fig. 1 of Marshall et al. [2021]

def from_pair24_csv(
    file_path: Path | str,
    fps: float,
) -> xr.Dataset:
    """The whole recording as one dataset on the session clock (frame i at i / fps)."""
    df = pd.read_csv(file_path)

    keypoint_names = [
        "HeadF",
        "HeadB",
        "HeadL",
        "SpineF",
        "SpineM",
        "SpineL",
        "Offset1",
        "Offset2",
        "HipL",
        "HipR",
        "ShoulderL",
        "ShoulderR",
    ]
    individual_names = ["an1", "an2"]
    position_types = ["aligned", "absolute"]
    n_frames = len(df)
    n_keypoints = len(keypoint_names)
    n_individuals = len(individual_names)
    n_space = 3

    position_array = np.zeros((n_frames, len(position_types), n_space, n_keypoints, n_individuals))
    confidence_array = np.ones((n_frames, len(position_types), n_keypoints, n_individuals))

    for p, pos_type in enumerate(position_types):
        csv_prefix = "alignedPosition" if pos_type == "aligned" else "absolutePosition"
        for i, individual in enumerate(individual_names):
            for j, keypoint in enumerate(keypoint_names):
                for k, coord in enumerate(["x", "y", "z"]):
                    col_name = f"{csv_prefix}_{individual}_{keypoint}_{coord}"
                    if col_name in df.columns:
                        position_array[:, p, k, j, i] = df[col_name].values

    time_coords = np.arange(n_frames, dtype=float) / fps

    ds = xr.Dataset(
        data_vars={
            "position": xr.DataArray(
                position_array,
                dims=["time", "position_type", "space", "keypoints", "individual"],
            ),
            "confidence": xr.DataArray(
                confidence_array,
                dims=["time", "position_type", "keypoints", "individual"],
            ),
        },
        coords={
            "time": time_coords,
            "position_type": position_types,
            "space": ["x", "y", "z"],
            "keypoints": keypoint_names,
            "individual": ["mouse 1", "mouse 2"],
        },
        attrs={"source_software": "DeepLabCut", "fps": fps},
    )

    com_data = np.zeros((n_frames, n_space, n_individuals))
    for i, individual in enumerate(individual_names):
        for j, coord in enumerate(["x", "y", "z"]):
            col_name = f"centerOfmass_{individual}_{coord}"
            if col_name in df.columns:
                com_data[:, j, i] = df[col_name].values
    ds["center_of_mass"] = xr.DataArray(com_data, dims=["time", "space", "individual"])

    ds["pairwise_distance"] = compute_norm(
        ds.center_of_mass.sel(individual="mouse 1") - ds.center_of_mass.sel(individual="mouse 2")
    )
    ds["nose_nose_distance"] = compute_norm(
        ds.position.sel(keypoints="HeadF", individual="mouse 1", position_type="absolute")
        - ds.position.sel(keypoints="HeadF", individual="mouse 2", position_type="absolute")
    )
    ds["velocity"] = compute_velocity(ds.position.sel(position_type="aligned"))
    ds["speed"] = compute_speed(ds.position.sel(position_type="aligned"))

    return ds


def trial_table_from_videos(
    video_dir: Path,
    cameras: list[str],
    n_frames: int,
    fps: float,
) -> pd.DataFrame:
    """One trial per video chunk; each file is named by its first frame on the session clock.

    A chunk that starts after the mocap's last frame has no data and is dropped, and the
    last trial ends where the mocap does. The timing goes into the alignment, which is
    the one source of trial timing: the GUI slices the long dataset by it.
    """
    starts = sorted(int(f.stem) for f in (video_dir / cameras[0]).glob("*.mp4"))
    starts = [s for s in starts if s < n_frames]
    stops = [min(s, n_frames) for s in starts[1:]] + [n_frames]

    rows = []
    for trial, (start, stop) in enumerate(zip(starts, stops), start=1):
        # Full path: pair_media stores the basename per trial and the path per stream.
        row = {"trial": trial, "start_time": start / fps, "stop_time": stop / fps}
        for cam in cameras:
            row[f"video_{cam}"] = str(video_dir / cam / f"{start}.mp4")
        rows.append(row)
    return pd.DataFrame(rows)
